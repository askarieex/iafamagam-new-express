BALANCE CALCULATION FUNCTION:
==============================

```javascript
async function getEffectiveBalance(accountId, ledgerHeadId, asOfDate) {
    // Get all transactions including corrections
    const transactions = await db.query(`
        SELECT amount, correction_type, original_period 
        FROM transaction_log 
        WHERE account_id = ? AND ledger_head_id = ?
        AND (
            (date <= ? AND original_period IS NULL) OR
            (original_period IS NOT NULL AND correction_relates_to_date <= ?)
        )
        ORDER BY log_id
    `, [accountId, ledgerHeadId, asOfDate, asOfDate]);
    
    let balance = 0;
    for (const tx of transactions) {
        balance += parseFloat(tx.amount);
    }
    
    return {
        effective_balance: balance,
        calculation_date: asOfDate,
        includes_corrections: transactions.some(tx => tx.correction_type)
    };
}
```

```javascript
async function createBalanceCorrection(originalLogId, correctAmount, currentAmount, reason) {
    const difference = currentAmount - correctAmount;
    
    if (difference === 0) {
        throw new Error('No correction needed - amounts are the same');
    }
    
    // Create correction entry
    const correctionEntry = {
        transaction_uuid: generateUUID(),
        amount: -difference, // Remove excess or add deficit
        description: `CORRECTION: ${reason}. Original: ${currentAmount}, Correct: ${correctAmount}, Adjustment: ${-difference}`,
        reference_transaction: originalLogId,
        correction_type: 'AMOUNT_ADJUSTMENT',
        date: new Date(), // Current date
        requires_approval: true,
        status: 'PENDING_APPROVAL'
    };
    
    return await db.TransactionLog.create(correctionEntry);
}
```
